# Four character rows used by the Name Entry screen.
#
# Uppercase/lowercase/symbol rows must resolve to exactly 26 selectable entries.
# The accent row may contain fewer; build_patch.py pads it to 26 with $80
# spaces. Those blank-looking cells are therefore selectable spaces and count
# toward the 9-character name limit.
#
# Supported named tokens:
#   <QUOTE_OPEN>  byte C3
#   <QUOTE_CLOSE> byte C4
#   <SPACE>       byte 80 (selectable space)
#   <GLYPH_CF>    original font glyph CF
#   <GLYPH_D0>    original font glyph D0
#   <GLYPH_D1>    original font glyph D1
#   <GLYPH_D2>    original font glyph D2
#
# Row framing (two leading blanks, one trailing blank and final $7F control)
# is generated automatically and is not authored here.

[uppercase]
ABCDEFGHIJKLMNOPQRSTUVWXYZ

[lowercase]
abcdefghijklmnopqrstuvwxyz

[symbols]
0123456789.,/'<QUOTE_OPEN><QUOTE_CLOSE>-<SPACE>!?()<GLYPH_CF><GLYPH_D0><GLYPH_D1><GLYPH_D2>

# Naming-safe canonical French range $D4-$E0.
# $E1-$E5 stay untouched because those slots are used by Name Entry graphics.
[accents]
Çàâçéèêëîïôùû
