# Generic rows used by the extended Name Entry screen.
#
# Each authored row must resolve to exactly 26 selectable entries. The builder
# renders exactly these three rows; localization components may add further
# rows without owning the generic 9-character/name-entry engine.
#
# Supported named tokens:
#   <QUOTE_OPEN>  byte C3
#   <QUOTE_CLOSE> byte C4
#   <SPACE>       byte 80 (selectable space)
#   <GLYPH_CF>    original font glyph CF
#   <GLYPH_D0>    original font glyph D0
#   <GLYPH_D1>    original font glyph D1
#   <GLYPH_D2>    original font glyph D2

[uppercase]
ABCDEFGHIJKLMNOPQRSTUVWXYZ

[lowercase]
abcdefghijklmnopqrstuvwxyz

[symbols]
0123456789.,/'<QUOTE_OPEN><QUOTE_CLOSE>-<SPACE>!?()<GLYPH_CF><GLYPH_D0><GLYPH_D1><GLYPH_D2>
